public static XMLGregorianCalendar getFormatoDate() {
    XMLGregorianCalendar todayGC = null;
    try {
        GregorianCalendar gCalendar = new GregorianCalendar();
        gCalendar.setTime(new Date());
        todayGC = DatatypeFactory.newInstance().newXMLGregorianCalendar(gCalendar);
    } catch (DatatypeConfigurationException de) {
        // manejo de excepción
    }
    return todayGC;
}
