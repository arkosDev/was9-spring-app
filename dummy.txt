@Autowired
    private AppServerHolder appSHolder;

    @Override
    @SuppressWarnings({"rawtypes", "unchecked"})
    public void consultaPolizaProgress(String numeroPoliza, RespuEmmaPolDto respuestaPolVO,
            Map<String, Boolean> filtros) throws ProcesException, AppServException, ChangeStatException {

        logger.info("***  ENTRA A EMMAPOLIZASZPROXY.CONSULTAPOLIZAPROGRESS  ***");
        /** Se obtiene un objeto con la conexion a progress */
        logger.info("consultaPolizaProgres - Conexion AppServer " + appSHolder.getAppServer());

        ConsultaAutos emma =
                EmmaPolUtil.connectToProgress(appSHolder.getAppServer(), ConsultaAutos.class);

        /** Definicion de vectores */
        Vector vectorAux = new Vector();
        Vector ttPoliza = new Vector();

        /** Definicion de resultSetHolders donde se almacenaran las respuestas de progress */
        ResultSetHolder ttConsPoliza = new ResultSetHolder();
        ResultSetHolder ttConsBeneficiario = new ResultSetHolder();
        ResultSetHolder ttConsNombreYO = new ResultSetHolder();

        ttPoliza.add(numeroPoliza);
        vectorAux.add(ttPoliza);

        try {
            emma.pstmconsultagralespolB2B(new Vector2ResultSet(vectorAux), ttConsPoliza,
                    ttConsBeneficiario, ttConsNombreYO);

            if (isFiltroActivo(filtros, KEY_CONSULTA_GENERALES)) {
                llenaVPoliza(ttConsPoliza, respuestaPolVO);
            } else {
                closeResultSet(ttConsPoliza,
                        "ERROR AL CERRAR EL RESULTSET AUXILIAR DE LA CONSULTA DE DATOS GENERALES");
            }

            if (isFiltroActivo(filtros, KEY_CONSULTA_BENEFICIARIO)) {
                llenaVBeneficiario(ttConsBeneficiario, respuestaPolVO);
            } else {
                closeResultSet(ttConsBeneficiario,
                        "ERROR AL CERRAR EL RESULTSET AUXILIAR DE LA CONSULTA DE BENEFICIARIOS");
            }

            if (isFiltroActivo(filtros, KEY_CONSULTA_CLIENTEYO)) {
                llenaVContratanteYO(ttConsNombreYO, respuestaPolVO);
            } else {
                closeResultSet(ttConsNombreYO,
                        "ERROR AL CERRAR EL RESULTSET AUXILIAR DE LA CONSULTA DE CLIENTESY/O");
            }

            if (isFiltroActivo(filtros, KEY_CONSULTA_CONDUCTORES_REGISTRADOS)) {
                logger.info("*** KEY_CONSULTA_CONDUCTORES_REGISTRADOS  *** ES TRUE");
                consultaConductoresProgress(numeroPoliza, respuestaPolVO);
            }

            if (isFiltroActivo(filtros, KEY_CONSULTA_BENEFICIARIOS_ACCIDENTES)) {
                logger.info("*** KEY_CONSULTA_BENEFICIARIOS_ACCIDENTES  *** ES TRUE");
                consultaBenefAccidentesProgress(numeroPoliza, respuestaPolVO);
            }

        } catch (RunTime4GLException e) {
            throw new ProcesException(
                    EjecutarMetodo + "emma.pstmconsultagralespolB2B -(RunTime4GLException) " + e.getMessage(),
                    e);
        } catch (SystemErrorException e) {
            throw new ProcesException(EjecutarMetodo
                    + "emma.pstmconsultagralespolB2B -(SystemErrorException) " + e.getMessage(), e);
        } catch (Open4GLException e) {
            throw new ProcesException(
                    EjecutarMetodo + "emma.pstmconsultagralespolB2B -(Open4GLException) " + e.getMessage(),
                    e);
        } catch (Exception e) {
            throw new ProcesException(
                    EjecutarMetodo + ": consultaPolizaProgress -(Exception) " + e.getMessage(), e);
        } finally {
            try {
                emma._release();
            } catch (Exception e) {
                logger.info(
                        " La conexion no pudo ser liberada manualmente, sera liberada por el pool de conexiones: "
                                + e.getMessage());
            }
        }
    }

    private boolean isFiltroActivo(Map<String, Boolean> filtros, String key) {
        return Optional.ofNullable(filtros)
                .map(map -> Boolean.TRUE.equals(map.get(key)))
                .orElse(false);
    }

    private void closeResultSet(ResultSetHolder holder, String errorMsg) {
        if (holder == null || holder.getResultSetValue() == null) {
            return;
        }
        try {
            holder.getResultSetValue().close();
        } catch (SQLException e) {
            logger.error(errorMsg, e);
        }
    }

    private void llenaVContratanteYO(ResultSetHolder ttConsNombreYO, RespuEmmaPolDto respuestaPolVO)

    private String getOptionalString(ResultSet rs, String columnName) {
        try {
            String value = rs.getString(columnName);
            return value != null ? value : "";
        } catch (SQLException ex) {
            return "";
        }
    }

    @SuppressWarnings({"rawtypes", "unchecked"})
    private void consultaConductoresProgress(String numeroPoliza, RespuEmmaPolDto respuestaPolVO)
            throws ProcesException, AppServException {
        ConsultaAutos emma = EmmaPolUtil.connectToProgress(appSHolder.getAppServer(), ConsultaAutos.class);
        Vector input = new Vector();
        Vector data = new Vector();
        ResultSetHolder holder = new ResultSetHolder();
        data.add(numeroPoliza);
        input.add(data);

        try {
            emma.pstmconsultacondB2B(new Vector2ResultSet(input), holder);
            // TODO: reemplazar Map por ConductorDto cuando esté disponible
            Vector result = new Vector(EmmaPolUtil.parseResultSet(holder, rs -> {
                java.util.Map<String, Object> row = new java.util.HashMap<>();
                row.put("chrNumPoliza", EmmaPolUtil.getString(rs, "chrNumPoliza"));
                row.put("chrNombre", getOptionalString(rs, "chrNombre"));
                return row;
            }));
            respuestaPolVO.setVConsConductores(result);
        } catch (RunTime4GLException e) {
            throw new ProcesException(
                    EjecutarMetodo + "emma.pstmconsultacondB2B -(RunTime4GLException) " + e.getMessage(), e);
        } catch (SystemErrorException e) {
            throw new ProcesException(
                    EjecutarMetodo + "emma.pstmconsultacondB2B -(SystemErrorException) " + e.getMessage(), e);
        } catch (Open4GLException e) {
            throw new ProcesException(
                    EjecutarMetodo + "emma.pstmconsultacondB2B -(Open4GLException) " + e.getMessage(), e);
        } catch (ResSetCallbException e) {
            throw new ProcesException(
                    "ERROR AL CONVERTIR DATOS llenaVOConductores -(ResSetCallbException) " + e.getMessage(), e);
        } finally {
            try {
                emma._release();
            } catch (Exception e) {
                logger.info(" La conexion no pudo ser liberada manualmente, sera liberada por el pool de conexiones: "
                        + e.getMessage());
            }
        }
    }
