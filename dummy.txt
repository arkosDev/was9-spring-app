public static List<String> sepConEsp(String texto, char separador) {
    if (null == texto || texto.isEmpty()) {
        return Collections.emptyList();
    }

    return Arrays.stream(texto.split(java.util.regex.Pattern.quote(String.valueOf(separador))))
        .map(String::trim)
        .filter(s -> !s.isEmpty())
        .collect(Collectors.toList());
}
