public static Document validaXMLconSAXValidator(String xml) throws InvDocException {

    SAXReader reader = new SAXReader(true);
    reader.setEncoding(ENCODE);
    reader.setErrorHandler(crearManejadorErrores());

    try (Reader stringReader = new StringReader(xml)) {
        Document docXmlRepositorio = reader.read(stringReader);

        reader.setFeature(SCHEMA, true);
        reader.setFeature(SCHEMA_CHECK, true);

        SAXValidator validator = new SAXValidator(reader.getXMLReader());
        validator.validate(docXmlRepositorio);

        return docXmlRepositorio;
    } catch (SAXException e) {
        throw new InvDocException(
            ConvertirXml + "(SAXException): " + e.getMessage());
    } catch (DocumentException e) {
        throw new InvDocException(
            CargarDatosValidacion + "(DocumentException): " + e.getMessage());
    } catch (IOException e) {
        throw new InvDocException(
            ConvertirXml + "(IOException): " + e.getMessage());
    }
}

private static XMLErrorHandler crearManejadorErrores() {
    return new XMLErrorHandler() {
        @Override
        public void error(SAXParseException exception) {
            Element errores = this.getErrors() != null
                ? this.getErrors()
                : DocumentHelper.createElement("errores");

            String strError = exception.getLocalizedMessage()
                .replaceAll("null", "")
                .replaceAll("faceta", "mascara");

            Element error = DocumentHelper.createElement("error");
            error.setText(strError);
            errores.add(error);

            this.setErrors(errores);
        }
    };
}
