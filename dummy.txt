    private static final String ENCODE = "ISO-8859-1";

    public String getXmlPoliza(final RespuEmmaPolDto respEmmDto)
        throws XMLStreamException, TransformerException, IOException {

        ByteArrayOutputStream xmlBytes = new ByteArrayOutputStream();
        XMLOutputFactory xof = XMLOutputFactory.newInstance();
        XMLStreamWriter hd = xof.createXMLStreamWriter(xmlBytes, ENCODE);

        hd.writeStartDocument(ENCODE, "1.0");
        hd.writeStartElement(POLIZA);

        addGenerales(respEmmDto, hd);
        addTitular(respEmmDto, hd);
        addCondicionesEmision(respEmmDto, hd);
        addBeneficiarioPreferente(respEmmDto, hd);
        addClienteYO(respEmmDto, hd);
        addConductoresRegistrados(respEmmDto, hd);
        addBeneficiariosAccidentes(respEmmDto, hd);
        addVehiculo(respEmmDto, hd);
        addEndosos(respEmmDto, hd);

        hd.writeEndElement(); // POLIZA
        hd.writeEndDocument();
        hd.flush();
        hd.close();

        String strXml = xmlBytes.toString(ENCODE);
        xmlBytes.close();

        Transformer transformer = TransformerFactory.newInstance().newTransformer();
        transformer.setOutputProperty(OutputKeys.INDENT, "yes");
        transformer.setOutputProperty("{http://xml.apache.org/xslt}indent-amount", "1");
        transformer.setOutputProperty(OutputKeys.ENCODING, ENCODE);
        StringWriter sw = new StringWriter();
        transformer.transform(new StreamSource(new StringReader(strXml)), new StreamResult(sw));
        strXml = sw.toString();
        return strXml;
    }

        hd.writeCharacters(assingSafeSring(recibosValueObject.getDcmImpuesto()).toCharArray(), 0,
            assingSafeSring(recibosValueObject.getDcmImpuesto()).length());
        hd.writeEndElement();
        hd.writeStartElement(PRIMATOTAL);
        hd.writeCharacters(assingSafeSring(recibosValueObject.getDcmPrimaTotal()).toCharArray(), 0,
            assingSafeSring(recibosValueObject.getDcmPrimaTotal()).length());
        hd.writeEndElement();

        hd.writeEndElement(); // RECIBO
    }

    private void addEquiposEspeciales(final RespuEmmaPolDto respEmmDto, XMLStreamWriter hd, String endoso)
        throws XMLStreamException {

        hd.writeStartElement(EQUIPO_ESP_ADAP);
        if (null != respEmmDto.getVConsEquipoEspecial() && !respEmmDto.getVConsEquipoEspecial().isEmpty()) {
            Iterator<EquipEspDto> iterEquipoEspecial = respEmmDto.getVConsEquipoEspecial().iterator();
            if (endoso != null && !endoso.equals("")) {
                while (iterEquipoEspecial.hasNext()) {
                    EquipEspDto equipoEspecialVO = iterEquipoEspecial.next();
                    if (equipoEspecialVO.getStrNumEndoso().equals(endoso)) {
                        addEquipoEspecial(hd, equipoEspecialVO);
                    }
                }
            } else {
                while (iterEquipoEspecial.hasNext()) {
                    EquipEspDto equipoEspecialVO = iterEquipoEspecial.next();
                    addEquipoEspecial(hd, equipoEspecialVO);
                }
            }
        } else {
            hd.writeStartElement(EQUIPO_ESPECIAL);
            hd.writeEndElement();
        }
        hd.writeEndElement(); // EQUIPO_ESP_ADAP
    }

    private void addMovimientos(final RespuEmmaPolDto respEmmDto, XMLStreamWriter hd, String endoso)
        throws XMLStreamException {

        hd.writeStartElement(MOVIMIENTOS);
        if (null != respEmmDto.getVConsEndososB() && !respEmmDto.getVConsEndososB().isEmpty()) {
            Iterator<EndososBDto> iterEndososB = respEmmDto.getVConsEndososB().iterator();
            if (endoso != null && !endoso.equals("")) {
                while (iterEndososB.hasNext()) {
                    EndososBDto endososBValueObject = iterEndososB.next();
                    if (endososBValueObject.getChrEndoso().equals(endoso)) {
                        addMovimiento(hd, endososBValueObject);
                    }
                }
            } else {
                while (iterEndososB.hasNext()) {
                    EndososBDto endososBValueObject = iterEndososB.next();
                    addMovimiento(hd, endososBValueObject);
                }
            }
        } else {
            hd.writeStartElement(MOVIMIENTO);
            hd.writeEndElement();
        }

        hd.writeEndElement(); // MOVIMIENTOS
    }

    private void addMovimiento(XMLStreamWriter hd, EndososBDto endososBValueObject)
        throws XMLStreamException {

        hd.writeStartElement(MOVIMIENTO);
        hd.writeStartElement(NUMEROPOLIZA);
        hd.writeCharacters(assingSafeSring(endososBValueObject.getChrPoliza()).toCharArray(), 0,
            assingSafeSring(endososBValueObject.getChrPoliza()).length());
        hd.writeEndElement();
        hd.writeStartElement(NUMERO_ENDOSO);
        hd.writeCharacters(assingSafeSring(endososBValueObject.getChrEndoso()).toCharArray(), 0,
            assingSafeSring(endososBValueObject.getChrEndoso()).length());
        hd.writeEndElement();
        hd.writeStartElement(DESCRIPCION);
        hd.writeCharacters(assingSafeSring(endososBValueObject.getChrDescEndoso()).toCharArray(), 0,
            assingSafeSring(endososBValueObject.getChrDescEndoso()).length());
        hd.writeEndElement();
        hd.writeStartElement(DATO_ANTERIOR);
        hd.writeCharacters(assingSafeSring(endososBValueObject.getChrDatoAnterior()).toCharArray(), 0,
            assingSafeSring(endososBValueObject.getChrDatoAnterior()).length());
        hd.writeEndElement();
        hd.writeStartElement(DATO_ACTUAL);
        hd.writeCharacters(assingSafeSring(endososBValueObject.getChrDatoActual()).toCharArray(), 0,
            assingSafeSring(endososBValueObject.getChrDatoActual()).length());
        hd.writeEndElement();
        hd.writeEndElement(); // MOVIMIENTO
    }

    private static String assingSafeSring(Object value) {
        if (value == null) {
            return "";
        }
        return value.toString();
    }
