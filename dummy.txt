private PaqEmiDto getValueObjectFromForm(PaqEmisionReq form) throws ProcessingException {
    try {
        return PaqEmiDto.builder()
                .nomPaq(form.getNomPaq())
                .idSubRamo(form.getSubRamFk())
                .dedDM(form.getDedDM())
                .dedRB(form.getDedRB())
                .dedRC(form.getDedRC())
                .sumRC(form.getSumRC())
                .sumRCPE(form.getSumRCPE())
                .sumGM(form.getSumGM())
                .sumEE(0)
                .sumAD(0)
                .sumACC(form.getSumACC())
                .sumAuSus(form.getSumAuSus())
                .sumProl(form.getSumProl())
                .valPrix(Boolean.TRUE.equals(form.getValPrix()))
                .extRC(Boolean.TRUE.equals(form.getExtRC()))
                .amplRC(Boolean.TRUE.equals(form.getAmplRC()))
                .dobRem(Boolean.TRUE.equals(form.getDobleRem()))
                .dedValCom(Boolean.TRUE.equals(form.getDedValCom()))
                .servAsist(Boolean.TRUE.equals(form.getServAsist()))
                .sumRCVia(Boolean.TRUE.equals(form.getBsumRCVia()))
                .rCExt(Boolean.TRUE.equals(form.getRCExt()))
                .build();
    } catch (Exception ex) {
        String errorMsg = "Error in PaquetesEmisionAction.getValueObjectFromForm(): " + ex;
        throw new ProcessingException(errorMsg, ex);
    }
}
