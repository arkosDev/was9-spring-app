package com.empresa.seguros.autos.serv.impl;

import static com.empresa.seguros.autos.util.ConstMsjError.EjecutarMetodo;
import static com.empresa.seguros.autos.util.Constantes.EMPTY;
import static com.empresa.seguros.autos.util.Constantes.SEPARADOR;

import com.empresa.seguros.autos.config.AppServerHolder;
import com.empresa.seguros.autos.exception.ProgException;
import com.empresa.seguros.autos.exception.ResSetCallbException;
import com.empresa.seguros.autos.serv.EmmaPdfProxyServ;
import com.progress.open4gl.ResultSetHolder;
import com.progress.open4gl.Vector2ResultSet;

import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.Vector;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import CotEmiAutos.EmisionAutos;

@Service
public class EmmaPdfProxyServImpl implements EmmaPdfProxyServ {

    private static final Logger logger = LoggerFactory.getLogger(EmmaPdfProxyServImpl.class);

    @Autowired
    AppServerHolder appSHolder;

    public Vector<Object> enviaImpresionEmmaProgress(Vector<Vector<Object>> ttDoctoImp)
            throws ProgException {

        EmisionAutos emma = null;
        Vector<Object> vectorRespuesta = null;
        ResultSetHolder ttRespuesta = new ResultSetHolder();

        try {
            emma = new EmisionAutos(appSHolder.getAppServer(), EMPTY, EMPTY, EMPTY);
            emma.pstmenviaimpind(new Vector2ResultSet(ttDoctoImp), ttRespuesta);

            vectorRespuesta = notifyResultSetProgressImprime(ttRespuesta);

        } catch (Exception e) {
            throw new ProgException(EjecutarMetodo + ": enviaImpresionEmmaProgress " + e.getMessage(), e);
        } finally {
            if (emma != null) {
                try {
                    emma._release();
                    emma = null;
                } catch (Exception e) {
                    logger.info("La conexión no pudo ser liberada manualmente, será liberada por el pool de conexiones: "
                            + e.getMessage());
                }
            }
        }

        return vectorRespuesta;
    }

    public Vector<Object> notifyResultSetProgressImprime(ResultSetHolder rsh)
            throws ResSetCallbException {

        logger.info("***En EmmaProxy::notifyResultSetProgressImprime");

        ResultSet rsCheck = rsh.getResultSetValue();
        if (rsCheck == null) {
            throw new ResSetCallbException(
                    "EmmaProxy:notifyResultSetProgressImprime - null ResultSet parameter.");
        }

        try (ResultSet rs = rsCheck) {

            Vector<Object> vectorRespuesta = new Vector<Object>();
            StringBuilder strError = new StringBuilder();

            while (rs.next()) {

                agregarCampo(rs, "chrNumPoliza", "ERRORES: EMA no regreso el numero de Poliza", false, strError, vectorRespuesta);
                agregarCampo(rs, "chrNumEndoso", "EMA no regreso el numero de Endoso", true, strError, vectorRespuesta);
                agregarCampo(rs, "chrSecuenEndo", "EMA no regreso el consecutivo del Endoso", true, strError, vectorRespuesta);
                agregarDescripcionError(rs, strError, vectorRespuesta);
            }

            if (vectorRespuesta.size() == 0) {
                vectorRespuesta.add("");
                vectorRespuesta.add("");
                vectorRespuesta.add("");
                vectorRespuesta.add("Error: EMA regresa respuesta vacia!!");
                logger.info("***En EmmaProxy::notifyResultSetProgressImprime");
            }

            return vectorRespuesta;

        } catch (SQLException e) {
            throw new ResSetCallbException(
                    "notifyResultSetProgressImprime: Error getting the result set Progress: " + e, e);
        }
    }

    private void agregarCampo(ResultSet rs, String columna, String mensajeError, boolean anteponerSeparador,
            StringBuilder strError, Vector<Object> vectorRespuesta) throws SQLException {

        String aux = rs.getString(columna);
        if (aux != null) {
            vectorRespuesta.add(aux);
        } else {
            if (anteponerSeparador) {
                strError.append(SEPARADOR);
            }
            strError.append(mensajeError);
            vectorRespuesta.add("");
        }
    }

    private void agregarDescripcionError(ResultSet rs, StringBuilder strError, Vector<Object> vectorRespuesta)
            throws SQLException {

        String aux = rs.getString("chrDescError");
        if (aux != null) {
            strError.append(SEPARADOR).append(aux);
            vectorRespuesta.add(aux);
        } else {
            vectorRespuesta.add(strError.length() == 0 ? "" : strError.toString());
        }
    }
}
