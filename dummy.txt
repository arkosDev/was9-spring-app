    public String getXmlPoliza() throws SAXException, IOException {

        OutputFormat of = new OutputFormat("XML", "ISO-8859-1", true);
        xmlBytes = new ByteArrayOutputStream();
        String strXml = null;
        of.setIndent(1);
        of.setIndenting(true);
        // of.setDoctype(null,"users.dtd");
        XMLSerializer serializer = new XMLSerializer(xmlBytes, of);

        ContentHandler hd = serializer.asContentHandler();

        hd.startDocument();
        hd.startElement("", "", ConstantesXMLSalida.POLIZA, new AttributesImpl());

        addGenerales(hd);
        addTitular(hd);
        addCondicionesEmision(hd);
        addBeneficiarioPreferente(hd);
        addClienteYO(hd);
        addConductoresRegistrados(hd);
        addBeneficiariosAccidentes(hd);
        addVehiculo(hd);
        addEndosos(hd);

        hd.endElement("", "", ConstantesXMLSalida.POLIZA);
        hd.endDocument();
        strXml = xmlBytes.toString();
        xmlBytes.close();
        xmlBytes = null;

        return strXml;
    }
