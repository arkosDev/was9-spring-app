if (null != ensDto.getBeneficiarios() && !ensDto.getBeneficiarios().isEmpty()) {
    logger.info("*** GUARDA BENEFICIARIOS PREFERENTES ***");
    try {
        List<BeneficiaroDto> beneficiarios = ensDto.getBeneficiarios().stream()
            .peek(benefDto -> {
                benefDto.setIdBenef(UUIDUtil.getUUIDLong());
                benefDto.setIdDatPol(ensDto.getIdDatPol());
                benefDto.setNumPol(ensDto.getNumPol());
                benefDto.setTipPers(ZERO);
            })
            .collect(Collectors.toList());
        benefSer.cargarBenef(beneficiarios);
    } catch (Exception e) {
        logger.info(
            "***En EmmaEndososProxy::createBeneficiario. Ocurrio un error al guardar el beneficiario Preferente: "
                + e.getMessage());
    }
}
