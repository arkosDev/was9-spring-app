private boolean validaNombreNoRepetido(final PaqEmiDto paqEmiDto) {
    try {
        final List<PaqEmiDto> paquz = paqRepo.findNomPaq(paqEmiDto.getNomPaq());
        return paquz.stream()
                .noneMatch(paqDto -> !paqEmiDto.getIdPaqEmi().equals(paqDto.getIdPaqEmi()));
    } catch (Exception ex) {
        logger.error(ex.getMessage());
        return false;
    }
}
