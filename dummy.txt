    private void armaError(String strErrores, XMLStreamWriter hd) throws Exception {

        try {
            hd.writeStartElement(ConstantesXMLSalida.ERRORES);

        } catch (XMLStreamException e) {

    /**
     * @param respuestas
     * @param idSolicitud
     * @return
     * @throws Exception
     */
    public String getXMLRespuestaEndososB(RespuestaEmmaValueObject[] respuestas, Long idSolicitud)
            throws Exception {
        logger.info("INICIA::ArmaXmlCancelacion:getXMLCancelacion([])");

        ByteArrayOutputStream xmlBytes = new ByteArrayOutputStream();
        XMLOutputFactory xof = XMLOutputFactory.newInstance();
        XMLStreamWriter hd = xof.createXMLStreamWriter(xmlBytes, ENCODE);
        String strXml = null;

        String respuestasId = "";
        String[] respuestasArray = null;
        String[] partesRespuestaArray = null;
        String numPoliza = "";
        List listaRespuestas = new ArrayList();

        try {
            hd.writeStartDocument(ENCODE, "1.0");
            hd.writeStartElement(ConstantesXMLEndosoB.ENDOSO_B);

            hd.writeStartElement(ConstantesXMLEndosoB.SOLICITUDID);
            hd.writeCharacters(assingSafeSring(idSolicitud));
            hd.writeEndElement();

            for (int i = 0; i < respuestas.length; i++) {
                respuestasId = "";

                if (!listaRespuestas.contains(respuestas[i].getStr_num_poliza())) {

                    if (respuestas[i].getChrDescripcion() != null
                            && !respuestas[i].getChrDescripcion().toString().equals("")) {
                        respuestasId = respuestasId.concat(respuestas[i].getChrDescripcion());
                    } else {
                        respuestasId = respuestasId.concat(
                                respuestas[i].getChrDocumentoEmitido().concat(ConstantesSolicitud.SEPARADOR));
                    }

                    numPoliza = respuestas[i].getStr_num_poliza();
                    listaRespuestas.add(respuestas[i].getStr_num_poliza());

                    respuestasArray = respuestasId.split("[|]");

                    for (int j = 0; j < respuestasArray.length; j++) {
                        hd.writeStartElement(ConstantesXMLEndosoB.ENDOSO);

                        hd.writeStartElement(ConstantesXMLCancelacion.NUMEROPOLIZA);
                        hd.writeCharacters(assingSafeSring(numPoliza));
                        hd.writeEndElement();

                        if (respuestasArray[j].toUpperCase().indexOf(ConstantesSolicitud.ERROR) != -1) {
                            armaError(respuestasArray[j], hd);
                        } else {
                            partesRespuestaArray = respuestasArray[j].split(",");

                            hd.writeStartElement(ConstantesXMLEndosoB.NUMEROENDOSO);
                            hd.writeCharacters(assingSafeSring(partesRespuestaArray[0]));
                            hd.writeEndElement();

                            hd.writeStartElement(ConstantesXMLEndosoB.CLAVE_TRABAJO);
                            hd.writeCharacters(assingSafeSring(partesRespuestaArray[1]));
                            hd.writeEndElement();

                            hd.writeStartElement(ConstantesXMLEndosoB.TIPO_ENDOSO);
                            hd.writeCharacters(assingSafeSring(partesRespuestaArray[2]));
                            hd.writeEndElement();
                        }

                        hd.writeEndElement(); // ENDOSO
                    }
                }
            }

            hd.writeEndElement(); // ENDOSO_B
            hd.writeEndDocument();
            hd.flush();

            strXml = xmlBytes.toString(ENCODE);
        } finally {
            hd.close();
            xmlBytes.close();
        }

        logger.info("TERMINA::ArmaXmlRespuestaSB:getXML([])");
        return strXml;
    }

    public Collection separarConEspacios(String strCadena, char strSeparador) {
        //antes static KMPS
        if (strCadena == null) {
            return new ArrayList();
        }
        Collection colSeparada = new ArrayList();
        char c;
        StringBuffer strValor = new StringBuffer();
        for (int i = 0; i < strCadena.length(); i++) {
            c = strCadena.charAt(i);
            for (int j = i + 1; c != strSeparador && j < strCadena.length(); j++, i++) {
                strValor.append(c);
                c = strCadena.charAt(j);
            }
            if (i + 1 == strCadena.length()) {
                strValor.append(c);
            }
            if (strValor.toString().trim().length() > 0) {
                colSeparada.add(strValor.toString().trim());
            }
            strValor = new StringBuffer();
        }
        // Elimina el ultimo separador de la ultima cadena
        int intColSize = colSeparada.size();
        if (intColSize > 0) {
            String strUltimoString = (String) ((List) colSeparada).get(intColSize - 1);
            if (strUltimoString.charAt(strUltimoString.length() - 1) == strSeparador) {
                strUltimoString = strUltimoString.substring(0, strUltimoString.length() - 1);
            }
            ((List) colSeparada).remove(intColSize - 1);
            ((List) colSeparada).add(intColSize - 1, strUltimoString);
        }
        return colSeparada;
    }
}

        StringTokenizer tokenizer = new StringTokenizer(sInput, sDelim);
        String sToken = null;
        String str[] = { "" };
        if (tokenizer != null && tokenizer.countTokens() > 0) {
            str = new String[tokenizer.countTokens()];
            int i = 0;
            while (tokenizer.hasMoreTokens()) {
                sToken = tokenizer.nextToken();
                if (sToken != null && sToken.length() > 0) {
                    str[i] = sToken;
                }
                i++;
            }
        }
        return (str);
    }
