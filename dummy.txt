    public String getXml(final String solicitante, final List<ProdCompDto> productos)
            throws Exception {

        // logger.info("INICIA::ArmaXmlPaquetesSB:getXml()");

        ByteArrayOutputStream xmlBytes = new ByteArrayOutputStream();
        XMLOutputFactory xof = XMLOutputFactory.newInstance();
        XMLStreamWriter hd = xof.createXMLStreamWriter(xmlBytes, ENCODE);

        String strXml;

        hd.writeStartDocument(ENCODE, "1.0");
        hd.writeStartElement(PRODUCTOS);

        // SOLICITANTE
        hd.writeStartElement(SOLICITANTE);
        hd.writeCharacters(TextUtil.valNulo(solicitante));
        hd.writeEndElement(); // SOLICITANTE

        for (int i = 0; i < productos.size(); i++) {

            final ProdCompDto product = productos.get(i);

            // PRODUCTOS
            hd.writeStartElement(PRODUCTO);

            // NOMBRE
            hd.writeStartElement(NOMBRE);
            hd.writeCharacters(TextUtil.valNulo(product.getNomProd()));
            hd.writeEndElement(); // NOMBRE

            // CLAVE
            hd.writeStartElement(CLAVE);
            hd.writeCharacters(TextUtil.valObjNulo(product.getProdId()));
            hd.writeEndElement(); // CLAVE

            // INICIO DE VIGENCIA
            hd.writeStartElement(INICIO_VIGENCIA);
            if (product.getFecVigIni() != null) {
                hd.writeCharacters(FechaUtil.format(product.getFecVigIni()));
            } else {
                hd.writeCharacters(EMPTY);
            }
            hd.writeEndElement(); // INICIO_VIGENCIA

            // FIN DE VIGENCIA
            hd.writeStartElement(FIN_VIGENCIA);
            if (product.getFecVigFin() != null) {
                hd.writeCharacters(FechaUtil.format(product.getFecVigFin()));
            } else {
                hd.writeCharacters(EMPTY);
            }
            hd.writeEndElement(); // FIN_VIGENCIA

            // TIPO POLIZA
            hd.writeStartElement(TIPOPOLIZA);
            if (product.getTipPol() != null) {
                String tipoPol = CatTipPolUtil.getKey(product.getTipPol());
                hd.writeCharacters(TextUtil.valNulo(tipoPol));
            } else {
                hd.writeCharacters(EMPTY);
            }
            hd.writeEndElement(); // TIPOPOLIZA

            // PRORRATEAR DERECHOS
            hd.writeStartElement(PRORRATEARDERECHOS);
            if (Boolean.valueOf(product.getProrDer())) {
                hd.writeCharacters(SI);
            } else {
                hd.writeCharacters(NO);
            }
            hd.writeEndElement(); // PRORRATEARDERECHOS

            // MODELOS VALIDOS DE
            hd.writeStartElement(MODELOSVALIDOSDE);
            hd.writeCharacters(TextUtil.valObjNulo(product.getModValInf()));
            hd.writeEndElement(); // MODELOSVALIDOSDE

            // MODELOS VALIDOS A
            hd.writeStartElement(MODELOSVALIDOSA);
            hd.writeCharacters(TextUtil.valObjNulo(product.getModValsup()));
            hd.writeEndElement(); // MODELOSVALIDOSA

            // FORMAS DE PAGO
            hd.writeStartElement(FORMAS_PAGO);
            generarFormasDePago(product.getFormPagVal(), hd);
            hd.writeEndElement(); // FORMAS_PAGO

            // VIGENCIAS
            hd.writeStartElement(VIGENCIAS);
            generarVigencias(product.getVigVal(), hd);
            hd.writeEndElement(); // VIGENCIAS

            // SERIES
            hd.writeStartElement(SERIES);

            // SERIES AUTOEXPEDIBLES
            hd.writeStartElement(SERIESAUTOEXPEDIBLES);
            generarSeries(product.getSerieAuto(), hd);
            hd.writeEndElement(); // SERIESAUTOEXPEDIBLES

            // SERIES EMISION
            hd.writeStartElement(SERIESEMISION);
            generarSeries(product.getSerPolEmi(), hd);
            hd.writeEndElement(); // SERIESEMISION

            hd.writeEndElement(); // SERIES

            // CONDICIONES EMISION
            hd.writeStartElement(CONDICIONESEMISION);
            obtenerCondicionEmision(product, hd);
            hd.writeEndElement(); // CONDICIONESEMISION

            // PAGO EN LINEA
            hd.writeStartElement(PAGOENLINEA);

            hd.writeStartElement(PAGOINMEDIATO);
            String valorPagoInmediato = obtenValorChecksPago(product.getPagInmVo());
            hd.writeCharacters(TextUtil.valNulo(valorPagoInmediato));
            hd.writeEndElement(); // PAGOINMEDIATO

            hd.writeStartElement(PAGOOBLIGATORIO);
            String valorPagoObligatorio = obtenValorChecksPago(product.getPagOblVo());
            // TODO: en el legacy este valor usaba el length() de valorPagoInmediato
            // en vez de valorPagoObligatorio; con writeCharacters(String) el bug
            // ya no aplica porque StAX no requiere length manual.
            hd.writeCharacters(TextUtil.valNulo(valorPagoObligatorio));
            hd.writeEndElement(); // PAGOOBLIGATORIO

            if (valorPagoInmediato.equals(SI)) {
                generaAfiliaciones(product.getAfilRela(), hd);
            }

            hd.writeEndElement(); // PAGOENLINEA

            // OBLIGATORIEDAD
            hd.writeStartElement(IS_OBLIGATORIO);
            if (product.getObligatorio()) {
                hd.writeCharacters(SI);
            } else {
                hd.writeCharacters(NO);
            }
            hd.writeEndElement(); // IS_OBLIGATORIO

            hd.writeEndElement(); // PRODUCTO
        }

        hd.writeEndElement(); // PRODUCTOS
        hd.writeEndDocument();
        hd.flush();
        hd.close();

        strXml = xmlBytes.toString(ENCODE);
        xmlBytes.close();

        // logger.info("TERMINA::ArmaXmlPaquetesSB:getXml()");

        return strXml;
    }

    private void obtenerCondicionEmision(ProdCompDto producto, XMLStreamWriter hd)
            throws Exception {

        if (producto.getCondEmision() == null) {
            return;
        }

        for (int i = 0; i < producto.getCondEmision().size(); i++) {

            hd.writeStartElement(CONDICION);

            final CondEmiDto condicion = producto.getCondEmision().get(i);

            // SUBRAMO
            hd.writeStartElement(SUBRAMO);
            if (condicion.getSubramo() != null) {
                String subramo = CatalogosUtil.getKeySubramo(condicion.getSubramo().toString());
                hd.writeCharacters(TextUtil.valNulo(subramo));
            } else {
                hd.writeCharacters(EMPTY);
            }
            hd.writeEndElement(); // SUBRAMO

            // TIPOS DE VEHICULO
            hd.writeStartElement(TIPOSVEHICULO);
            generarTiposVehiculo(condicion.getValTip(), hd, condicion.getSubramo().toString());
            hd.writeEndElement(); // TIPOSVEHICULO

            // DEDUCIBLES
            hd.writeStartElement(DEDUCIBLES);
            generarDeducibles(condicion, hd);
            hd.writeEndElement(); // DEDUCIBLES

            // SUMAS ASEGURADAS
            hd.writeStartElement(SUMASASEGURADAS);
            generarSumasCondicionesEmision(condicion, hd);
            hd.writeEndElement(); // SUMASASEGURADAS

            generarValorConvenidoComercial(condicion, hd);
            generarLogicos(condicion, hd);

            // TIPOS DE CARGA
            hd.writeStartElement(TIPOSCARGA);
            generarTiposCarga(condicion.getTipCarga(), hd);
            hd.writeEndElement(); // TIPOSCARGA

            // TARIFAS
            hd.writeStartElement(TARIFAS);
            generarTarifas(condicion.getTarifVal(), hd);
            hd.writeEndElement(); // TARIFAS

            // PAQUETES DE EMISION
            hd.writeStartElement(PAQUETEEMISION);

            if (condicion.getPaquetes() != null) {
                for (int j = 0; j < condicion.getPaquetes().size(); j++) {
                    // BUG LEGACY preservado: el original indexa con "i" (índice del loop
                    // externo de condiciones) en vez de "j" (índice del loop de paquetes).
                    // Se deja igual por ser migración 1:1; avisar si se debe corregir.
                    final PaqEmiDto paquete = condicion.getPaquetes().get(i);
                    generarPaquetesEmision(paquete, hd);
                }
            }

            hd.writeEndElement(); // PAQUETEEMISION
            hd.writeEndElement(); // CONDICION
        }
    }

    private void generarTiposVehiculo(String tiposVehiculo, XMLStreamWriter hd, String subramo)
            throws XMLStreamException {

        if (tiposVehiculo == null || tiposVehiculo.trim().equals(EMPTY)) {
            return;
        }
        String[] tipoVehiculoIndividual = tiposVehiculo.split("[|]");
        for (int i = 0; i < tipoVehiculoIndividual.length; i++) {
            if (tipoVehiculoIndividual[i] != null && !tipoVehiculoIndividual[i].trim().equals(EMPTY)) {
                String relacionSubRamo = CatalogosUtil.getSubramtipVeh(subramo);
                if (relacionSubRamo != null) {
                    String[] relacionesSubRamo = relacionSubRamo.split("[|]");
                    for (int j = 0; j < relacionesSubRamo.length; j++) {
                        if (relacionesSubRamo[j].equals(tipoVehiculoIndividual[i])) {
                            hd.writeStartElement(TIPOVEHICULO);
                            String tipoCatalogo = CatalogosUtil.getKeySubramTipVeh(tipoVehiculoIndividual[i]);
                            hd.writeCharacters(TextUtil.valNulo(tipoCatalogo));
                            hd.writeEndElement(); // TIPOVEHICULO
                        }
                    }
                }
            }
        }
    }

    private void generarTarifas(String tarifas, XMLStreamWriter hd) throws XMLStreamException {
        if (tarifas == null || tarifas.trim().equals(EMPTY)) {
            return;
        }
        String[] tarifaIndividual = tarifas.split("[|]");
        for (int i = 0; i < tarifaIndividual.length; i++) {
            if (tarifaIndividual[i] != null && !tarifaIndividual[i].trim().equals(EMPTY)) {
                hd.writeStartElement(TARIFA);
                hd.writeCharacters(TextUtil.valNulo(tarifaIndividual[i]));
                hd.writeEndElement(); // TARIFA
            }
        }
    }

    private void generarTiposCarga(String strTiposCarga, XMLStreamWriter hd) throws XMLStreamException {

    }

        for (int i = 0; i < serieIndividual.length; i++) {
            if (serieIndividual[i] != null && !serieIndividual[i].trim().equals(EMPTY)) {
                hd.writeStartElement(SERIEPOLIZA);
                hd.writeCharacters(TextUtil.valNulo(serieIndividual[i]));
                hd.writeEndElement(); // SERIEPOLIZA
            }
        }
    }

    private void generarVigencias(String vigencias, XMLStreamWriter hd) throws XMLStreamException {
        if (vigencias == null || vigencias.trim().equals(EMPTY)) {
            return;
        }
        String[] vigencia = vigencias.split("[|]");
        for (int i = 0; i < vigencia.length; i++) {
            if (vigencia[i] != null && !vigencia[i].trim().equals(EMPTY)) {
                hd.writeStartElement(VIGENCIA);
                String vigenciaLeyenda = CatalogosUtil.getKeyVigencias(vigencia[i]);
                hd.writeCharacters(TextUtil.valNulo(vigenciaLeyenda));
                hd.writeEndElement(); // VIGENCIA
            }
        }
    }

    private void generarFormasDePago(String formasPago, XMLStreamWriter hd) throws XMLStreamException {
        if (formasPago == null || formasPago.trim().equals(EMPTY)) {
            return;
        }
        String[] formaPago = formasPago.split("[|]");
        for (int i = 0; i < formaPago.length; i++) {
            String[] claveFormaPago = formaPago[i].split(",");
            if (claveFormaPago[0] != null && !claveFormaPago[0].trim().equals(EMPTY)) {
                String formaPagoLeyenda = CatalogosUtil.getKeyFormaPago(claveFormaPago[0]);
                if (formaPagoLeyenda != null && !formaPagoLeyenda.equals(EMPTY)) {
                    hd.writeStartElement(FORMAPAGO);
                    hd.writeCharacters(TextUtil.valNulo(formaPagoLeyenda));
                    hd.writeEndElement(); // FORMAPAGO
                }
            }
        }
    }

    private static void getSumas(PaqEmiDto paquete, XMLStreamWriter hd) throws Exception {
        DecimalFormat decimalFormat = new DecimalFormat("#0");
        String equipoEspecial;
        String adaptaciones;
        try {
            hd.writeStartElement(SUMASASEGURADAS);

            hd.writeStartElement(RCBIENES);
            hd.writeCharacters(TextUtil.valObjNulo(paquete.getSumRC()));
            hd.writeEndElement(); // RCBIENES

            hd.writeStartElement(GASTOSMEDICOS);
            hd.writeCharacters(TextUtil.valObjNulo(paquete.getSumGM()));
            hd.writeEndElement(); // GASTOSMEDICOS

            // EQUIPO ESPECIAL
            if (paquete.getDblSumEE() != null) {
                equipoEspecial = decimalFormat.format(paquete.getDblSumEE().doubleValue());
            } else {
                equipoEspecial = decimalFormat.format(new Double(0).doubleValue());
            }
            hd.writeStartElement(EQUIPOESPECIAL);
            hd.writeCharacters(TextUtil.valNulo(equipoEspecial));
            hd.writeEndElement(); // EQUIPOESPECIAL

            // ADAPTACIONES
            if (paquete.getSumAD() != null) {
                adaptaciones = decimalFormat.format(paquete.getSumAD().doubleValue());
            } else {
                adaptaciones = decimalFormat.format(new Double(0).doubleValue());
            }
            hd.writeStartElement(ADAPTACIONES);
            hd.writeCharacters(TextUtil.valNulo(adaptaciones));
            hd.writeEndElement(); // ADAPTACIONES

            hd.writeStartElement(ACCIDENTES);
            hd.writeCharacters(TextUtil.valObjNulo(paquete.getSumACC()));
            hd.writeEndElement(); // ACCIDENTES

            hd.writeStartElement(AUTOSUSTITUTO);
            hd.writeCharacters(TextUtil.valObjNulo(paquete.getSumAuSus()));
            hd.writeEndElement(); // AUTOSUSTITUTO

            hd.writeStartElement(FIANZA);
            hd.writeCharacters(TextUtil.valObjNulo(paquete.getSumProl()));
            hd.writeEndElement(); // FIANZA

            hd.writeEndElement(); // SUMASASEGURADAS
        } catch (XMLStreamException e) {
            throw new Exception(ERROR_AL_GENERAR_XML, e);
        }
    }

    public String obtenValorChecksPago(String check) {
        String valor = EMPTY;
        if (check != null && check.equals("1")) {
            valor = SI;

        }
        return valor;
    }
