public List<SerieDto> findSerProdTip(List<String> prodIds, List<String> tipos) {
    
    if (prodIds == null || prodIds.isEmpty() || tipos == null || tipos.isEmpty()) {
        return Collections.emptyList();
    }

    String prodPlaceholders = prodIds.stream()
            .map(p -> "?")
            .collect(Collectors.joining(","));

    String tiposPlaceholders = tipos.stream()
            .map(t -> "?")
            .collect(Collectors.joining(","));

    String sql = """
        SELECT 
            SERIE.lng_producto_id,
            SERIE.str_tipo_serie,
            LISTAGG(SERIE.str_serie_poliza, '|') 
                WITHIN GROUP (ORDER BY SERIE.str_serie_poliza) AS series
        FROM EMA_SERIES SERIE
        WHERE SERIE.lng_producto_id IN (%s)
          AND SERIE.str_tipo_serie IN (%s)
        GROUP BY SERIE.lng_producto_id, SERIE.str_tipo_serie
        """.formatted(prodPlaceholders, tiposPlaceholders);

    // Versión moderna con PreparedStatementSetter
    return jdbcTemplate.query(sql, 
        ps -> {
            int index = 1;
            for (String id : prodIds) {
                ps.setString(index++, id);
            }
            for (String tipo : tipos) {
                ps.setString(index++, tipo);
            }
        },
        (rs, rowNum) -> SerieDto.builder()
            .idProd(rs.getLong("lng_producto_id"))
            .tipSer(rs.getString("str_tipo_serie"))
            .serPol(rs.getString("series"))
            .build()
    );
}
