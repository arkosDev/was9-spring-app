public class DateAdapter extends XmlAdapter<String, Timestamp> {

    private static final String FORMAT = "yyyy-MM-dd HH:mm:ss";

    @Override
    public Timestamp unmarshal(String v) throws Exception {
        return Timestamp.valueOf(v);
    }

    @Override
    public String marshal(Timestamp v) throws Exception {
        if (v == null) return null;
        return new SimpleDateFormat(FORMAT).format(v);
    }
}

@XmlElement(name = "fechaSolicitud")
@XmlJavaTypeAdapter(DateAdapter.class)
private Timestamp fecha;

import javax.xml.bind.annotation.adapters.XmlJavaTypeAdapter;

public class MapAdapter extends XmlAdapter<MapAdapter.MapElements, Map<String, Integer>> {

    public static class MapElements {
        @XmlElement(name = "entrada")
        public List<MapEntry> entries = new ArrayList<>();
    }

    public static class MapEntry {
        @XmlElement(name = "clave")
        public String key;
        @XmlElement(name = "valor")
        public Integer value;

        public MapEntry() {}
        public MapEntry(String key, Integer value) {
            this.key   = key;
            this.value = value;
        }
    }

    @Override
    public Map<String, Integer> unmarshal(MapElements v) {
        Map<String, Integer> map = new LinkedHashMap<>();
        for (MapEntry e : v.entries) map.put(e.key, e.value);
        return map;
    }

    @Override
    public MapElements marshal(Map<String, Integer> v) {
        MapElements elements = new MapElements();
        if (v != null) {
            for (Map.Entry<String, Integer> e : v.entrySet()) {
                elements.entries.add(new MapEntry(e.getKey(), e.getValue()));
            }
        }
        return elements;
    }
}

@XmlElement(name = "estatusSolicitud")
@XmlJavaTypeAdapter(MapAdapter.class)
private Map<String, Integer> estatus;
